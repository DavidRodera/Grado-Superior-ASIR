% Macro para insertar imágenes de forma segura tanto dentro como fuera de grids
\providecommand{\gridimg}[1]{%
  \begin{tcolorbox}[blankest]
    \begin{tikzpicture}
      \node[inner sep=0pt, clip] (img) {\includegraphics[width=\linewidth]{#1}};
      \draw[grisLinea, line width=0.5pt] (img.south west) rectangle (img.north east);
    \end{tikzpicture}
  \end{tcolorbox}%
}

% ── Portada Académica ─────────────────────────────────────────────
\newcommand{\portada}[6]{%
  \def\footercustom{#6}%
  \newgeometry{margin=2.5cm}%
  \thispagestyle{portada}%
  \setstretch{0.85}%
  \vspace*{2cm}
  \begin{center}
    {\montserratfont\large\bfseries\color{colorTeal}\MakeUppercase{#1}}\par\vspace{0.6em}
    {\montserratfont\Huge\bfseries\color{azulProfundo}#2}\par\vspace{0.8em}
    {\montserratfont\large\color{grisCabecera}#3}\par\vspace{1.8em}
    \begin{tikzpicture}
      \draw[colorTeal, line width=3.5pt] (0,0) -- (6.5,0);
    \end{tikzpicture}\par\vspace{1.8em}
    {\montserratfont\normalsize\color{black}#4\\[0.5em]#5}
  \end{center}
  \vfill
  \clearpage
  \restoregeometry
  \pagecolor{white}
  \color{grisTexto}
}
