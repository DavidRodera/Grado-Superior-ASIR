% ── Portada Premium Sólida ─────────────────────────────────────────
\newcommand{\portada}[3]{%
  \newgeometry{margin=0pt}%
  \thispagestyle{portada}%
  \begin{tikzpicture}[remember picture,overlay]
    % Fondo sólido
    \fill[portadaBG]
      (current page.south west) rectangle (current page.north east);

    \fill[portadaStripe]
      ([xshift=-2cm]current page.north east)
      rectangle (current page.south east);
    \node[rotate=90, anchor=center, text=blancoSuave, font=\cabinfont\small]
      at ([xshift=-1cm]current page.east) {\MakeUppercase{#2}};
    \node (titulo) [anchor=north west, xshift=2.5cm, yshift=-9.5cm, text width=13cm, align=left]
      at (current page.north west) {%
        \color{blancoPuro}%
        \montserratfont\fontsize{46}{64}\selectfont\bfseries
        \MakeUppercase{#1}\par
      };
    \draw[color=blancoSuave, line width=0.4pt]
      ([yshift=-0.8cm]titulo.south west)
      -- ([xshift=-2.8cm, yshift=-0.8cm]titulo.south west -| current page.north east);
    \node[anchor=south west, xshift=2.5cm, yshift=1.8cm,
      text=blancoSuave, font=\montserratfont\normalsize]
      at (current page.south west) {#3};
  \end{tikzpicture}%
  \clearpage
  \restoregeometry%
  \pagecolor{white}%
  \color{grisTexto}%
}
